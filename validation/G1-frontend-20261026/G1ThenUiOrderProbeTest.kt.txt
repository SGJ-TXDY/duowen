package cn.embeddedquiz

import org.junit.runner.Description
import org.junit.runner.RunWith
import org.junit.runner.Runner
import org.junit.runner.manipulation.Filter
import org.junit.runners.Suite
import org.robolectric.RobolectricTestRunner

/** Isolated-copy control only, never committed to app/src: fixes G1 -> UI order explicitly. */
private fun g1ThenUi(): List<Runner> {
    fun one(type: Class<*>, method: String): Runner = RobolectricTestRunner(type).apply {
        filter(Filter.matchMethodDescription(Description.createTestDescription(type, method)))
    }
    return listOf(
        one(ReviewReminderTest::class.java, "notificationReviewEntryWinsAfterUnfinishedExamIsRestored"),
        one(PracticeEntriesUiTest::class.java, "homeEntriesOpenBanksAndTabBarOffersExam"),
    )
}

class G1ThenUiRunner(testClass: Class<*>) : Suite(testClass, g1ThenUi())

@RunWith(G1ThenUiRunner::class)
class G1ThenUiOrderProbeTest
